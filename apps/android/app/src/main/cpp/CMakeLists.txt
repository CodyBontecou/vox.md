cmake_minimum_required(VERSION 3.22)
project(vox_intelligence LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_CURL LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER GGML_OPENMP GGML_NATIVE GGML_LLAMAFILE GGML_METAL GGML_BLAS GGML_BACKEND_DL GGML_AVX GGML_AVX2 GGML_AVX512 GGML_FMA GGML_F16C GGML_SSE42)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_library(vox_intelligence SHARED intelligence.cpp)
target_link_libraries(vox_intelligence PRIVATE llama)
target_link_options(vox_intelligence PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
